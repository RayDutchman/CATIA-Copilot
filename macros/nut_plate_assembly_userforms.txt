' ============================================================
' [1/3] 以下代码粘贴到 FlipForm 的代码模块
' ============================================================

Private Sub btnFlip_Click()
    If Not g_loopActive Then Exit Sub
    If Not g_hasLastComp Then Exit Sub
    ' 直接翻转并重装，不中断 SelectElement2
    g_flipDir = Not g_flipDir
    Call MoveLastNutPlate
    CATIA.StatusBar = "已翻转方向，继续选孔..."
End Sub

Private Sub btnSwap_Click()
    If Not g_loopActive Then Exit Sub
    If Not g_hasLastComp Then Exit Sub
    ' 直接对调铆钉孔并重装，不中断 SelectElement2
    g_swapHoles = Not g_swapHoles
    Call MoveLastNutPlate
    CATIA.StatusBar = "已对调铆钉孔，继续选孔..."
End Sub

Private Sub btnUndo_Click()
    If Not g_loopActive Then Exit Sub
    If g_historyCount <= 0 Then Exit Sub
    ' 直接撤销，不中断 SelectElement2
    Call UndoLastNutPlate
End Sub

Private Sub btnStop_Click()
    g_loopActive = False
    g_hasLastComp = False
    Unload Me
    Call SendEsc
End Sub


' ============================================================
' [2/3] 以下代码粘贴到 SwitchToPartForm 的代码模块
' ============================================================

Private Sub btnContinue_Click()
    ' 若 ActiveDocument 不是 PartDocument，ContinueSelectNutPlate 会提示并返回，
    ' 窗体保持打开，用户可再次切换后重试。
    Call ContinueSelectNutPlate
End Sub

Private Sub btnCancel_Click()
    Unload SwitchToPartForm
End Sub


' ============================================================
' [3/3] 以下代码粘贴到 SwitchDocForm 的代码模块
' ============================================================

Private Sub btnContinue_Click()
    ' 若 ActiveDocument 不是 ProductDocument，ContinueSelectProduct 会提示并返回。
    Call ContinueSelectProduct
End Sub

Private Sub btnCancel_Click()
    Unload SwitchDocForm
End Sub
